cmake_minimum_required(VERSION 3.29)
project(D2RRItemDatabase VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(ITEMDB_PLUGIN "Build Windows loader plugin and preview" ON)
set(SDK_DIR "${CMAKE_SOURCE_DIR}/.deps/sdk" CACHE PATH "Pinned D2RLoader SDK")
set(JSON_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/.deps/json" CACHE PATH "nlohmann include root")
include(CTest)
add_library(itemdb_core src/database.cpp src/query.cpp src/state.cpp src/prototype.cpp)
target_include_directories(itemdb_core PUBLIC include "${JSON_INCLUDE_DIR}")
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
  set_property(TARGET itemdb_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(itemdb_cli src/cli.cpp)
target_link_libraries(itemdb_cli PRIVATE itemdb_core)
if(BUILD_TESTING)
  add_executable(itemdb_tests tests/core_tests.cpp)
  target_link_libraries(itemdb_tests PRIVATE itemdb_core)
  add_test(NAME core COMMAND itemdb_tests "${CMAKE_SOURCE_DIR}/tests/fixture.json")
  add_executable(itemdb_ui_tests tests/prototype_tests.cpp)
  target_link_libraries(itemdb_ui_tests PRIVATE itemdb_core)
  add_test(NAME prototype COMMAND itemdb_ui_tests "${CMAKE_SOURCE_DIR}/data/database.json")
endif()
if(ITEMDB_PLUGIN AND WIN32)
  enable_language(RC)
  # Only consume the public headers; SDK tests/examples are independent projects.
  add_library(itemdb_sdk INTERFACE)
  target_include_directories(itemdb_sdk INTERFACE "${SDK_DIR}/include")
  add_library(itemdb_plugin SHARED src/plugin.cpp src/plugin.rc)
  target_link_libraries(itemdb_plugin PRIVATE itemdb_core itemdb_sdk)
  set_target_properties(itemdb_plugin PROPERTIES OUTPUT_NAME "d2rl-item-database" PREFIX "")
  install(TARGETS itemdb_plugin RUNTIME DESTINATION d2rloader/plugins)
  install(FILES data/database.json DESTINATION d2rloader/plugins/item-database)
endif()
if(MSVC)
  foreach(target itemdb_cli itemdb_tests itemdb_ui_tests itemdb_plugin)
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  endforeach()
endif()
